```c run id=enum-and-switch
#include <stdbool.h>
#include <stdio.h>

enum state { START, SIGN, INT, DOT, FRAC, DEAD };

static enum state step(enum state s, char ch) {
    bool digit = ch >= '0' && ch <= '9';
    switch (s) {
    case START: return ch == '+' || ch == '-' ? SIGN : digit ? INT : DEAD;
    case SIGN: return digit ? INT : DEAD;
    case INT: return digit ? INT : ch == '.' ? DOT : DEAD;
    case DOT:
    case FRAC: return digit ? FRAC : DEAD;
    case DEAD: return DEAD;
    }
    return DEAD;
}

static bool accepts(const char *text) {
    enum state s = START;
    for (const char *p = text; *p; p++) s = step(s, *p);
    return s == INT || s == FRAC;
}

int main(void) {
    const char *inputs[] = {"42", "-3.5", "1.", "+", "007", ".5", "4.2.1"};
    for (size_t i = 0; i < sizeof inputs / sizeof inputs[0]; i++)
        printf("%s %s\n", inputs[i], accepts(inputs[i]) ? "true" : "false");
    return 0;
}
```

- Verdict: idiom. C names the states with an `enum`, and a `switch` on the state gives the transitions. Compilers warn with `-Wswitch` when an enum value has no case. Lexers generated by tools such as lex and re2c are C programs of this shape, or with the table in an array.
